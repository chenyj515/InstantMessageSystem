cmake_minimum_required(VERSION 3.16)
project(InstantMessageSystem VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB IMS_SOURCES CONFIGURE_DEPENDS "src/*.cpp")
add_library(ims_core ${IMS_SOURCES})
target_include_directories(ims_core PUBLIC include)
if(MSVC)
  target_compile_options(ims_core PUBLIC /utf-8 /W4)
else()
  target_compile_options(ims_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
add_executable(InstantMessageSystem main.cpp)
target_link_libraries(InstantMessageSystem PRIVATE ims_core)
if(MINGW)
  target_link_options(InstantMessageSystem PRIVATE -municode)
endif()
option(IMS_STATIC_RUNTIME "Build standalone MinGW executables" ON)
if(MINGW AND IMS_STATIC_RUNTIME)
  target_link_options(InstantMessageSystem PRIVATE -static -static-libgcc -static-libstdc++)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(ims_tests tests/core_tests.cpp)
  target_link_libraries(ims_tests PRIVATE ims_core)
  if(MINGW AND IMS_STATIC_RUNTIME)
    target_link_options(ims_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME core_business_rules COMMAND ims_tests)
endif()
